cmake_minimum_required(VERSION 3.16)
project(portfolio_cybr_light LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(CYBR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
add_executable(cybr-light "${CYBR_ROOT}/cybr-light/src/main.cpp")
add_executable(cybr-numerical "${CYBR_ROOT}/cybr-light/tests/numerical.cpp")
add_executable(cybr-coating-tests coating-tests.cpp)
foreach(target cybr-light cybr-numerical cybr-coating-tests)
  target_include_directories(${target} PRIVATE "${CYBR_ROOT}/cybr-light/include")
  if(MSVC)
    target_compile_definitions(${target} PRIVATE NOMINMAX _USE_MATH_DEFINES)
    target_compile_options(${target} PRIVATE /O2 /EHsc /openmp)
  else()
    target_compile_options(${target} PRIVATE -O3)
  endif()
endforeach()
